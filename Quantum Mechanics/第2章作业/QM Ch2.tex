\section{}
设粒子限制在矩形匣子中运动，即
\[
V(x,y,z)=
\begin{cases}
0, & 0<x<a,\ 0<y<b,\ 0<z<c,\\
\infty, & \text{其余区域}.
\end{cases}
\]
求粒子的能量本征值和本征波函数。如 $a=b=c$，讨论能级的简并度。

\vspace{0.5cm}

\noindent\textbf{解：}在势阱内部 $0<x<a,\ 0<y<b,\ 0<z<c$ 时，$V(x,y,z)=0$；在边界及阱外，$V=\infty$，因此波函数必须满足边界条件
\[
\psi(0,y,z)=\psi(a,y,z)=0,
\]
\[
\psi(x,0,z)=\psi(x,b,z)=0,
\]
\[
\psi(x,y,0)=\psi(x,y,c)=0.
\]

在阱内，定态薛定谔方程为
\[
-\frac{\hbar^2}{2m}\nabla^2\psi(x,y,z)=E\psi(x,y,z),
\]
即
\[
\pdv[2]{\psi}{x}+\pdv[2]{\psi}{y}+\pdv[2]{\psi}{z}+\frac{2mE}{\hbar^2}\psi=0.
\]

令波函数分离变量：
\[
\psi(x,y,z)=X(x)Y(y)Z(z).
\]
代入方程并除以 $XYZ$，得
\[
\frac{X''(x)}{X(x)}+\frac{Y''(y)}{Y(y)}+\frac{Z''(z)}{Z(z)}=-\frac{2mE}{\hbar^2}.
\]
由于三项分别只依赖于不同变量，故可分别令
\[
\frac{X''}{X}=-k_x^2,\qquad \frac{Y''}{Y}=-k_y^2,\qquad \frac{Z''}{Z}=-k_z^2,
\]
并有
\[
k_x^2+k_y^2+k_z^2=\frac{2mE}{\hbar^2}.
\]
于是三个方向分别满足
\[
X''+k_x^2X=0,\qquad Y''+k_y^2Y=0,\qquad Z''+k_z^2Z=0.
\]

结合边界条件
\[
X(0)=X(a)=0,\qquad Y(0)=Y(b)=0,\qquad Z(0)=Z(c)=0,
\]
可知只允许正弦形式的非平凡解：
\[
X_{n_x}(x)=A_x\sin\left(\frac{n_x\pi x}{a}\right),
\]
\[
Y_{n_y}(y)=A_y\sin\left(\frac{n_y\pi y}{b}\right),
\]
\[
Z_{n_z}(z)=A_z\sin\left(\frac{n_z\pi z}{c}\right),
\]
其中
\[
n_x,n_y,n_z=1,2,3,\dots
\]
且
\[
k_x=\frac{n_x\pi}{a},\qquad k_y=\frac{n_y\pi}{b},\qquad k_z=\frac{n_z\pi}{c}.
\]

因此能量本征值为
\[
E_{n_xn_yn_z}=\frac{\hbar^2}{2m}\left[\left(\frac{n_x\pi}{a}\right)^2+\left(\frac{n_y\pi}{b}\right)^2+\left(\frac{n_z\pi}{c}\right)^2\right]
=\frac{\pi^2\hbar^2}{2m}\left(\frac{n_x^2}{a^2}+\frac{n_y^2}{b^2}+\frac{n_z^2}{c^2}\right).
\]

再由归一化条件
\[
\int_0^a \abs{X_{n_x}(x)}^2\,\dd x=1,\qquad \int_0^b \abs{Y_{n_y}(y)}^2\,\dd y=1,\qquad \int_0^c \abs{Z_{n_z}(z)}^2\,\dd z=1,
\]
并利用
\[
\int_0^a \sin^2\left(\frac{n_x\pi x}{a}\right)\,\dd x=\frac{a}{2},
\]
同理对 $y,z$ 方向也成立，可得
\[
A_x=\sqrt{\frac{2}{a}},\qquad A_y=\sqrt{\frac{2}{b}},\qquad A_z=\sqrt{\frac{2}{c}}.
\]
故归一化本征波函数为
\[
\psi_{n_xn_yn_z}(x,y,z)=\sqrt{\frac{8}{abc}}\sin\left(\frac{n_x\pi x}{a}\right)\sin\left(\frac{n_y\pi y}{b}\right)\sin\left(\frac{n_z\pi z}{c}\right),
\]
其中 $0<x<a,\ 0<y<b,\ 0<z<c$；阱外 $\psi=0$。

当 $a=b=c\equiv L$ 时，能量退化为
\[
E_{n_xn_yn_z}=\frac{\pi^2\hbar^2}{2mL^2}\left( n_x^2+n_y^2+n_z^2 \right).
\]
此时能量仅依赖于 $n_x^2+n_y^2+n_z^2$，因此凡是满足相同平方和的不同三元组 $(n_x,n_y,n_z)$ 都对应同一能级，从而出现简并。

简并度的讨论如下：
\begin{itemize}
  \item 若 $n_x,n_y,n_z$ 三个量两两不同，例如 $(1,2,3)$，则其所有排列对应 $6$ 个不同本征态，因此该能级为 $6$ 重简并；
  \item 若其中两个相同、一个不同，例如 $(1,1,2)$，则不同排列有 $3$ 个，因此该能级为 $3$ 重简并；
  \item 若三个都相同，例如 $(1,1,1)$，则只有 $1$ 个本征态，因此该能级非简并。
\end{itemize}

此外，还可能出现偶然简并，即并非简单排列关系的不同三元组也满足相同的平方和；这时总简并度应等于所有满足
\[
n_x^2+n_y^2+n_z^2=\text{常数}
\]
的正整数三元组个数。

\section{}
设粒子处于一维无限深方势阱中，
\[
V(x)=
\begin{cases}
0, & 0<x<a,\\
\infty, & x<0,\ x>a.
\end{cases}
\]
证明处于能量本征态 $\psi_n(x)$ 的粒子，
\[
\overline{x}=a/2, \qquad \overline{(x-\overline{x})^2}=\frac{a^2}{12}\left(1-\frac{6}{n^2\pi^2}\right).
\]
讨论 $n\to\infty$ 的情况，并与经典力学计算结果比较。

\vspace{0.5cm}

\noindent\textbf{证明：}
一维无限深方势阱的归一化能量本征函数为
\[
\psi_n(x)=\sqrt{\frac{2}{a}}\sin\left(\frac{n\pi x}{a}\right),\qquad n=1,2,3,\dots,
\]
阱外 $\psi_n(x)=0$，故
\[
\abs{\psi_n(x)}^2=\frac{2}{a}\sin^2\left(\frac{n\pi x}{a}\right).
\]

先计算$\overline{x}$：
\begin{align*}
\overline{x}
&=\ev{x} \\
&=\int_0^a x\abs{\psi_n(x)}^2\,\dd x \\
&=\frac{2}{a}\int_0^a x\sin^2\left(\frac{n\pi x}{a}\right)\,\dd x \\
&=\frac{1}{a}\int_0^a x\left[1-\cos\left(\frac{2n\pi x}{a}\right)\right]\,\dd x \\
&=\frac{1}{a}\int_0^a x\,\dd x-\frac{1}{a}\int_0^a x\cos\left(\frac{2n\pi x}{a}\right)\,\dd x \\
&=\frac{1}{a}\eval{\frac{x^2}{2}}_0^a-\frac{1}{a}\eval{\frac{x\sin\left(\frac{2n\pi x}{a}\right)}{\frac{2n\pi}{a}}+\frac{\cos\left(\frac{2n\pi x}{a}\right)}{\left(\frac{2n\pi}{a}\right)^2}}_0^a \\
&=\frac{a}{2}.
\end{align*}

对于方差，先计算$\ev{x^2}$：
\begin{align*}
\ev{x^2}
&=\int_0^a x^2\abs{\psi_n(x)}^2\,\dd x \\
&=\frac{2}{a}\int_0^a x^2\sin^2\left(\frac{n\pi x}{a}\right)\,\dd x \\
&=\frac{1}{a}\int_0^a x^2\left[1-\cos\left(\frac{2n\pi x}{a}\right)\right]\,\dd x \\
&=\frac{1}{a}\int_0^a x^2\,\dd x-\frac{1}{a}\int_0^a x^2\cos\left(\frac{2n\pi x}{a}\right)\,\dd x \\
&=\frac{1}{a}\eval{\frac{x^3}{3}}_0^a
-\frac{1}{a}\eval{\frac{x^2\sin\left(\frac{2n\pi x}{a}\right)}{\frac{2n\pi}{a}}
+\frac{2x\cos\left(\frac{2n\pi x}{a}\right)}{\left(\frac{2n\pi}{a}\right)^2}
-\frac{2\sin\left(\frac{2n\pi x}{a}\right)}{\left(\frac{2n\pi}{a}\right)^3}}_0^a \\
&=\frac{a^2}{3}-\frac{a^2}{2n^2\pi^2}.
\end{align*}

因此：
\begin{align*}
\overline{\left(x-\overline{x}\right)^2}
&=\ev{x^2}-\ev{x}^2 \\
&=\left(\frac{a^2}{3}-\frac{a^2}{2n^2\pi^2}\right)-\left(\frac{a}{2}\right)^2 \\
&=\frac{a^2}{12}\left(1-\frac{6}{n^2\pi^2}\right).
\end{align*}

当 $n\to\infty$ 时，
\[
\overline{\left(x-\overline{x}\right)^2}\to\frac{a^2}{12}.
\]

经典情形下，区域$(0,a)$中的粒子被击中的概率密度是$1/a$，故
\[
\ev{x}_{\mathrm{cl}}
=\frac{1}{a}\int_0^a x\,\dd x
=\frac{a}{2},
\]
\[
\ev{x^2}_{\mathrm{cl}}
=\frac{1}{a}\int_0^a x^2\,\dd x
=\frac{a^2}{3},
\]
所以
\[
\overline{\left(x-\overline{x}\right)^2}_{\mathrm{cl}}
=\ev{x^2}_{\mathrm{cl}}-\ev{x}_{\mathrm{cl}}^2
=\frac{a^2}{12}.
\]
因此当 $n\to\infty$ 时，量子结果趋于经典结果。

\section{}
设粒子处于一维无限深方势阱中，
\[
V(x)=
\begin{cases}
0, & |x|<a/2,\\
\infty, & |x|>a/2.
\end{cases}
\]
处于基态 $(n=1$，见 2.2 节式(12)$)$。求粒子的动量分布。

\vspace{0.5cm}

\noindent\textbf{解：}基态波函数为
\[
\psi_1(x)=
\begin{cases}
\sqrt{\dfrac{2}{a}}\cos\left(\dfrac{\pi x}{a}\right), & \abs{x}<\dfrac{a}{2},\\
0, & \abs{x}>\dfrac{a}{2}.
\end{cases}
\]

动量表象波函数为
\[
\varphi_1(p)=\frac{1}{\sqrt{2\pi\hbar}}\int_{-\infty}^{+\infty}\psi_1(x)e^{-ipx/\hbar}\, \dd x.
\]
因此
\begin{align*}
  \varphi_1(p)&=\frac{1}{\sqrt{2\pi\hbar}}\sqrt{\frac{2}{a}}\int_{-a/2}^{a/2}\cos\left(\frac{\pi x}{a}\right)e^{-ipx/\hbar}\, \dd x\\
  &=\frac{1}{\sqrt{2\pi\hbar}}\sqrt{\frac{2}{a}}\int_{-a/2}^{a/2}\cos\left(\frac{\pi x}{a}\right)\cos\left(\frac{px}{\hbar}\right)\, \dd x\\
  &=\frac{1}{\sqrt{2\pi\hbar}}\sqrt{\frac{2}{a}}\cdot\frac{1}{2}
  \int_{-a/2}^{a/2}
  \left[
  \cos\left(\left(\frac{\pi}{a}-\frac{p}{\hbar}\right)x\right)
  +\cos\left(\left(\frac{\pi}{a}+\frac{p}{\hbar}\right)x\right)
  \right]\, \dd x \\
  &=\frac{1}{\sqrt{2\pi\hbar}}\sqrt{\frac{2}{a}}
  \left[
  \frac{\sin\left[\left(\frac{\pi}{a}-\frac{p}{\hbar}\right)\frac{a}{2}\right]}{\frac{\pi}{a}-\frac{p}{\hbar}}
  +
  \frac{\sin\left[\left(\frac{\pi}{a}+\frac{p}{\hbar}\right)\frac{a}{2}\right]}{\frac{\pi}{a}+\frac{p}{\hbar}}
  \right] \\
  &=\frac{1}{\sqrt{2\pi\hbar}}\sqrt{\frac{2}{a}}
  \cos\left(\frac{\frac{p}{\hbar}a}{2}\right)
  \left[
  \frac{1}{\frac{\pi}{a}-\frac{p}{\hbar}}
  +
  \frac{1}{\frac{\pi}{a}+\frac{p}{\hbar}}
  \right] \\
  &=\frac{1}{\sqrt{2\pi\hbar}}\sqrt{\frac{2}{a}}
  \cos\left(\frac{\frac{p}{\hbar}a}{2}\right)
  \frac{2\pi/a}{\left(\frac{\pi}{a}\right)^2-\frac{p}{\hbar}^2} \\
  &=\sqrt{\frac{a}{\pi\hbar}}\,
  \frac{2\pi\cos\left(\frac{pa}{2\hbar}\right)}{\pi^2-\left(\frac{pa}{\hbar}\right)^2}.
\end{align*}

故动量分布为
\[
w(p)=\abs{\varphi_1(p)}^2
=\frac{4a}{\pi\hbar}
\frac{\pi^2\cos^2\left(\frac{pa}{2\hbar}\right)}
{\left[\pi^2-\left(\frac{pa}{\hbar}\right)^2\right]^2}.
\]

即
\[
w(p)\, \dd p
=
\frac{4a}{\pi\hbar}
\frac{\pi^2\cos^2\left(\frac{pa}{2\hbar}\right)}
{\left[\pi^2-\left(\frac{pa}{\hbar}\right)^2\right]^2}
\, \dd p
\]
表示粒子动量落在 \((p,p+\dd p)\) 内的概率。

\section{}
设粒子处于无限深方势阱
\[
V(x)=
\begin{cases}
0, & 0<x<a,\\
\infty, & x<0,\ x>a
\end{cases}
\]
中，粒子波函数为 $\psi(x)=Ax(x-a)$，$A$ 为归一化常数。
\begin{enumerate}[label=(\alph*)]
  \item 求 $A$；
  \item 求测得粒子处于能量本征态
  \[
  \psi_n(x)=\sqrt{\frac{2}{a}}\sin\frac{n\pi x}{a}
  \]
  的概率 $P_n$，特别是 $P_1$。
\end{enumerate}
提示：用 $\psi_n(x)$ 展开，$\psi(x)=\sum_n C_n\psi_n(x)$，$P_n=|C_n|^2$。

\begin{enumerate}[label=(\alph*),resume]
  \item 作图，比较 $\psi(x)$ 与 $\psi_1(x)$ 曲线。从 $P_1\gg P_n\,(n\ne 1)$ 来说明两条曲线非常相似，即 $\psi(x)$ 几乎与基态 $\psi_1(x)$ 完全相同。
\end{enumerate}

\vspace{0.5cm}

\noindent\textbf{解：}

\begin{enumerate}[label=(\alph*)]
  \item 由归一化条件
\begin{align*}
1
&=\int_0^a \abs{\psi(x)}^2\, \dd x \\
&=A^2\int_0^a x^2(x-a)^2\, \dd x \\
&=A^2\int_0^a \left(x^4-2ax^3+a^2x^2\right)\, \dd x \\
&=A^2\eval{\left(\frac{x^5}{5}-\frac{ax^4}{2}+\frac{a^2x^3}{3}\right)}_0^a \\
&=A^2a^5\left(\frac{1}{5}-\frac{1}{2}+\frac{1}{3}\right) \\
&=\frac{A^2a^5}{30}.
\end{align*}
故
\[
A=\sqrt{\frac{{30}}{a^{5}}}.
\]

  \item 将 \(\psi(x)\) 按本征函数展开：
\[
\psi(x)=\sum_{n=1}^{\infty}C_n\psi_n(x),
\qquad
C_n=\int_0^a \psi_n(x)\psi(x)\, \dd x.
\]
\[
C_n=\sqrt{\frac{2}{a}}A\int_0^a x(x-a)\sin\left(\frac{n\pi x}{a}\right)\, \dd x.
\]
其中
\begin{align*}
\int_0^a x(x-a)\sin\left(\frac{n\pi x}{a}\right)\, \dd x
&=\int_0^a \left(x^2-ax\right)\sin\left(\frac{n\pi x}{a}\right)\, \dd x \\
&=\frac{2a^3}{n^3\pi^3}\left[(-1)^n-1\right].
\end{align*}
故
\[
C_n=\sqrt{\frac{2}{a}}A\cdot \frac{2a^3}{n^3\pi^3}\left[(-1)^n-1\right]=\frac{4\sqrt{15}}{n^3\pi^3}\left[(-1)^n-1\right].
\]

\[
P_n=\abs{C_n}^2
=\frac{240}{n^6\pi^6}\left[(-1)^n-1\right]^2=
\begin{cases}
0, & n\ \text{为偶数},\\[6pt]
\dfrac{960}{n^6\pi^6}, & n\ \text{为奇数}.
\end{cases}
\]
\[
\implies P_1=\frac{960}{\pi^6}.
\]

注意到 $P_1\approx 0.999 \approx 1$，而归一化条件要求
\[
\sum_{n=1}^\infty \abs{C_n}^2 = \sum_{n=1}^\infty P_n = 1,
\]
因此$P_1 \gg P_n$，两条曲线应非常相似。

  \item 实线为$\psi_1(x)$，虚线为$\psi(x)$。
\begin{center}
  \begin{tikzpicture}[x=1.2cm,y=4cm]
  \def\a{6}

  \draw[->] (-0.4,0) -- (\a+0.5,0) node[right] {$x$};
  \draw[->] (0,-0.15) -- (0,0.8) node[above] {$\psi$};

  \draw[dashed] (0,0) -- (0,-0.08);
  \draw[dashed] (\a,0) -- (\a,-0.08);
  \node[below left] at (0,0) {$0$};
  \node[below left] at (\a,0) {$a$};
  \node[below] at ({\a/2},0) {$\dfrac{a}{2}$};

  \draw[domain=0:\a,smooth,variable=\x,thick]
    plot ({\x},{0.58*sin(180*\x/\a)});

  \draw[domain=0:\a,smooth,variable=\x,thick,dashed]
    plot ({\x},{0.56*(4*\x/\a)*(1-\x/\a)});

  \fill (0,0) circle (0.7pt);
  \fill (\a,0) circle (0.7pt);
  \end{tikzpicture}
\end{center}
\end{enumerate}

\section{}
同上题。设粒子处于基态 $(n=1)$，
\[
E_1=\frac{\pi^2\hbar^2}{2ma^2}.
\]
设 $t=0$ 时刻阱宽突然变为 $2a$，粒子波函数来不及改变，即
\[
\psi(x,0)=\psi_1(x)=\sqrt{\frac{2}{a}}\sin\frac{\pi x}{a}.
\]
试问：对于加宽了的无限深方势阱
\[
V(x)=
\begin{cases}
0, & 0<x<2a,\\
\infty, & x<0,\ x>2a
\end{cases}
\]
$\psi(x,0)$ 是否还是能量本征态？求测得粒子能量仍为 $E_1$ 的概率。（答：$1/2$）

提示：阱宽为 $2a$ 的无限深方势阱中的粒子能量本征值
\[
\varepsilon_n=\frac{n^2\pi^2\hbar^2}{8ma^2},
\]
本征函数
\[
\psi_n(x)=\sqrt{\frac{1}{a}}\sin\frac{n\pi x}{2a}.
\]
$\varepsilon_2=E_1$；$\psi(x,0)$ 用 $\psi_n(x)$ 展开，$\psi(x,0)=\sum_n C_n\psi_n(x)$，求出 $|C_2|^2$。

\vspace{0.5cm}
\noindent\textbf{解：}
阱宽突变后，粒子波函数仍为
\[
\psi(x,0)=
\begin{cases}
\sqrt{\dfrac{2}{a}}\sin\left(\dfrac{\pi x}{a}\right), & 0<x<a,\\
0, & a<x<2a.
\end{cases}
\]
新的无限深方势阱 \(0<x<2a\) 的本征函数与本征值为
\[
\psi_n(x)=\sqrt{\frac{1}{a}}\sin\left(\frac{n\pi x}{2a}\right),
\qquad
\varepsilon_n=\frac{n^2\pi^2\hbar^2}{8ma^2}.
\]
由于
\[
\varepsilon_2=\frac{4\pi^2\hbar^2}{8ma^2}=\frac{\pi^2\hbar^2}{2ma^2}=E_1,
\]
故测得粒子能量仍为 \(E_1\) 的概率为
\[
P(E_1)=\abs{C_2}^2,
\]
其中
\begin{align*}
C_2
&=\int_0^{2a}\psi_2(x)\psi(x,0)\, \dd x \\
&=\int_0^a \sqrt{\frac{1}{a}}\sin\left(\frac{2\pi x}{2a}\right)\sqrt{\frac{2}{a}}\sin\left(\frac{\pi x}{a}\right)\, \dd x \\
&=\frac{\sqrt{2}}{a}\int_0^a \sin^2\left(\frac{\pi x}{a}\right)\, \dd x \\
&=\frac{\sqrt{2}}{a}\cdot \frac{a}{2} \\
&=\frac{1}{\sqrt{2}}.
\end{align*}

因此
\[
P(E_1)=\abs{C_2}^2=\frac{1}{2}.
\]
所以，\(\psi(x,0)\) 不是加宽后势阱的能量本征态；测得粒子能量仍为 \(E_1\) 的概率为$\frac{1}{2}$。

\section{}
设粒子（能量 $E>0$）从左入射，碰到图 2.15 所示的势阱，求透射系数与反射系数。

\begin{figure}[H]
  \centering
  \includegraphics[width = 0.3 \textwidth]{2.6.png}
\end{figure}

\vspace{0.5cm}

\noindent\textbf{解：}
设势能为
\[
V(x)=
\begin{cases}
-V_0, & x<0,\\
0, & x>0.
\end{cases}
\]
粒子自左向右入射，且 \(E>0\)。

两侧波数分别为
\[
k_1=\frac{\sqrt{2m\left(E+V_0\right)}}{\hbar},
\qquad
k_2=\frac{\sqrt{2mE}}{\hbar}.
\]

于是
\[
\psi(x)=
\begin{cases}
Ae^{ik_1x}+Be^{-ik_1x}, & x<0,\\
Ce^{ik_2x}, & x>0.
\end{cases}
\]

在 \(x=0\) 处，波函数与其一阶导数连续：
\[
A+B=C,
\]
\[
ik_1\left(A-B\right)=ik_2C.
\]
解得：
\[
\frac{B}{A}=\frac{k_1-k_2}{k_1+k_2}.
\]
\[
\frac{C}{A}
=
1+\frac{k_1-k_2}{k_1+k_2}
=
\frac{2k_1}{k_1+k_2}.
\]

故反射系数为
\[
R=\abs{\frac{B}{A}}^2
=
\left(\frac{k_1-k_2}{k_1+k_2}\right)^2.
\]
透射系数为
\[
T=\frac{j_{\mathrm{t}}}{j_{\mathrm{i}}}
=\frac{k_2}{k_1}\abs{\frac{C}{A}}^2
=\frac{k_2}{k_1}\left(\frac{2k_1}{k_1+k_2}\right)^2
=\frac{4k_1k_2}{\left(k_1+k_2\right)^2}.
\]
其中
\[
k_1=\frac{\sqrt{2m\left(E+V_0\right)}}{\hbar},
\qquad
k_2=\frac{\sqrt{2mE}}{\hbar}.
\]

\section{}
利用 Hermite 多项式的递推关系（附录 A3，式(13)），证明谐振子波函数满足下列关系
\[
x\psi_n(x)=\frac{1}{\alpha}\left[\sqrt{\frac{n}{2}}\,\psi_{n-1}(x)+\sqrt{\frac{n+1}{2}}\,\psi_{n+1}(x)\right]
\]
\[
x^2\psi_n(x)=\frac{1}{2\alpha^2}\left[\sqrt{n(n-1)}\,\psi_{n-2}(x)+(2n+1)\psi_n(x)+\sqrt{(n+1)(n+2)}\,\psi_{n+2}(x)\right].
\]
并由此证明，在 $\psi_n$ 态下，$\overline{x}=0$，$\overline{V}=E_n/2$。

\vspace{0.5cm}

\noindent\textbf{证明：}
谐振子归一化波函数可写为
\[
\psi_n(x)=N_n H_n(\xi)e^{-\xi^2/2},
\qquad
\xi=\alpha x,
\]
其中
\[
N_n=\left(\frac{\alpha}{\sqrt{\pi}\,2^n n!}\right)^{1/2}.
\]
Hermite 多项式的递推关系为
\[
2\xi H_n(\xi)=2nH_{n-1}(\xi)+H_{n+1}(\xi).
\]

于是
\[
x\psi_n(x)=\frac{\xi}{\alpha}\psi_n(x)
=\frac{N_n}{2\alpha}\left[2nH_{n-1}(\xi)+H_{n+1}(\xi)\right]e^{-\xi^2/2}.
\]
又由于
\[
\psi_{n-1}(x)=N_{n-1}H_{n-1}(\xi)e^{-\xi^2/2},
\qquad
\psi_{n+1}(x)=N_{n+1}H_{n+1}(\xi)e^{-\xi^2/2},
\]
并且
\[
\frac{nN_n}{N_{n-1}}=\sqrt{\frac{n}{2}},
\qquad
\frac{N_n}{2N_{n+1}}=\sqrt{\frac{n+1}{2}},
\]
故
\[
x\psi_n(x)=\frac{1}{\alpha}\left[\sqrt{\frac{n}{2}}\,\psi_{n-1}(x)+\sqrt{\frac{n+1}{2}}\,\psi_{n+1}(x)\right].
\]

再对上式再乘一个 \(x\)，得
\[
x^2\psi_n(x)
=
\frac{1}{\alpha}\sqrt{\frac{n}{2}}\,x\psi_{n-1}(x)
+
\frac{1}{\alpha}\sqrt{\frac{n+1}{2}}\,x\psi_{n+1}(x).
\]
分别代入前一结果：
\[
x\psi_{n-1}(x)
=
\frac{1}{\alpha}\left[\sqrt{\frac{n-1}{2}}\,\psi_{n-2}(x)+\sqrt{\frac{n}{2}}\,\psi_n(x)\right],
\]
\[
x\psi_{n+1}(x)
=
\frac{1}{\alpha}\left[\sqrt{\frac{n+1}{2}}\,\psi_n(x)+\sqrt{\frac{n+2}{2}}\,\psi_{n+2}(x)\right].
\]
因此
\[
x^2\psi_n(x)=\frac{1}{2\alpha^2}\left[\sqrt{n(n-1)}\,\psi_{n-2}(x)+(2n+1)\psi_n(x)+\sqrt{(n+1)(n+2)}\,\psi_{n+2}(x)\right].
\]

对第一式左乘 \(\psi_n^*(x)\) 并积分，得
\[
\overline{x}=\ev{x}
=
\frac{1}{\alpha}\left[
\sqrt{\frac{n}{2}}\int_{-\infty}^{+\infty}\psi_n^*(x)\psi_{n-1}(x)\, \dd x
+
\sqrt{\frac{n+1}{2}}\int_{-\infty}^{+\infty}\psi_n^*(x)\psi_{n+1}(x)\, \dd x
\right]
=0.
\]
再对第二式左乘 \(\psi_n^*(x)\) 并积分，得
\begin{align*}
\ev{x^2}
&=
\frac{1}{2\alpha^2}\Bigl[
\sqrt{n(n-1)}\int_{-\infty}^{+\infty}\psi_n^*(x)\psi_{n-2}(x)\,\dd x \\
&\qquad\qquad
+ (2n+1)\int_{-\infty}^{+\infty}\psi_n^*(x)\psi_n(x)\,\dd x \\
&\qquad\qquad
+ \sqrt{(n+1)(n+2)}\int_{-\infty}^{+\infty}\psi_n^*(x)\psi_{n+2}(x)\,\dd x
\Bigr] \\
&= \frac{2n+1}{2\alpha^2}.
\end{align*}

谐振子势能为
\[
V(x)=\frac{1}{2}m\omega^2x^2,
\]
故
\[
\overline{V}=\frac{1}{2}m\omega^2\ev{x^2}
=\frac{m\omega^2}{2}\cdot\frac{2n+1}{2\alpha^2}
=\frac{m\omega^2}{2}\cdot\frac{2n+1}{2}\cdot\frac{\hbar}{m\omega}
=\frac{2n+1}{4}\hbar\omega.
\]
而
\[
E_n=\left(n+\frac{1}{2}\right)\hbar\omega=\frac{2n+1}{2}\hbar\omega,
\]
所以
\[
\overline{V}=\frac{E_n}{2}.
\]

故证得
\[
\overline{x}=0,
\qquad
\overline{V}=\frac{E_n}{2}.
\]

\section{}
同上题，利用 Hermite 多项式的求导公式（附录 A3，式(14)），证明
\[
\frac{d}{dx}\psi_n(x)=\alpha\left[\sqrt{\frac{n}{2}}\,\psi_{n-1}-\sqrt{\frac{n+1}{2}}\,\psi_{n+1}\right]
\]
\[
\frac{d^2}{dx^2}\psi_n(x)=\frac{\alpha^2}{2}\left[\sqrt{n(n-1)}\,\psi_{n-2}-(2n+1)\psi_n+\sqrt{(n+1)(n+2)}\,\psi_{n+2}\right].
\]
并由此证明，在 $\psi_n$ 态下，$\overline{p}=0$，$\overline{T}=\overline{p^2}/2m=E_n/2$。

\vspace{0.5cm}

\noindent\textbf{证明：}
谐振子归一化波函数为
\[
\psi_n(x)=N_n H_n(\xi)e^{-\xi^2/2},
\qquad
\xi=\alpha x,
\]
其中
\[
N_n=\left(\frac{\alpha}{\sqrt{\pi}\,2^n n!}\right)^{1/2}.
\]
Hermite 多项式的求导公式为
\[
\dv{H_n}{\xi}=2nH_{n-1}(\xi),
\]

因此：
\begin{align*}
\dv{\psi_n}{x}
&=\alpha N_n \dv{\xi}\left[ H_n(\xi)e^{-\xi^2/2} \right] \\
&=\alpha N_n \left[ \dv{H_n}{\xi}-\xi H_n(\xi) \right] e^{-\xi^2/2} \\
&=\alpha N_n \left[ 2nH_{n-1}(\xi)-\xi H_n(\xi) \right] e^{-\xi^2/2} \\
&=\alpha N_n \left[ 2nH_{n-1}(\xi)-\frac{1}{2}\left( 2nH_{n-1}(\xi)+H_{n+1}(\xi) \right) \right] e^{-\xi^2/2} \\
&=\alpha N_n \left[ nH_{n-1}(\xi)-\frac{1}{2}H_{n+1}(\xi) \right] e^{-\xi^2/2} \\
&=\alpha\left[ \frac{nN_n}{N_{n-1}}\psi_{n-1}(x)-\frac{N_n}{2N_{n+1}}\psi_{n+1}(x) \right] \\
&=\alpha\left[ \sqrt{\frac{n}{2}}\,\psi_{n-1}(x)-\sqrt{\frac{n+1}{2}}\,\psi_{n+1}(x) \right].
\end{align*}

再对上式求导，
\[
\dv[2]{\psi_n}{x}
=
\alpha\sqrt{\frac{n}{2}}\dv{\psi_{n-1}}{x}
-\alpha\sqrt{\frac{n+1}{2}}\dv{\psi_{n+1}}{x}.
\]
这里
\[
\dv{\psi_{n-1}}{x}
=
\alpha\left[ \sqrt{\frac{n-1}{2}}\,\psi_{n-2}(x)-\sqrt{\frac{n}{2}}\,\psi_n(x) \right],
\]
\[
\dv{\psi_{n+1}}{x}
=
\alpha\left[ \sqrt{\frac{n+1}{2}}\,\psi_n(x)-\sqrt{\frac{n+2}{2}}\,\psi_{n+2}(x) \right].
\]
故
\[
\dv[2]{\psi_n}{x}
=
\frac{\alpha^2}{2}\left[
\sqrt{n(n-1)}\,\psi_{n-2}(x)
-(2n+1)\psi_n(x)
+\sqrt{(n+1)(n+2)}\,\psi_{n+2}(x)
\right].
\]

动量算符为
\[
\hat{p}=-i\hbar \dv{}{x}.
\]
于是
\[
\overline{p}
=
\int_{-\infty}^{+\infty}\psi_n^*(x)\hat{p}\psi_n(x)\, \dd x
=
-i\hbar\int_{-\infty}^{+\infty}\psi_n^*(x)\dv{\psi_n}{x}\, \dd x.
\]
代入上式，
\[
\overline{p}
=
-i\hbar\alpha\left[
\sqrt{\frac{n}{2}}\int_{-\infty}^{+\infty}\psi_n^*(x)\psi_{n-1}(x)\, \dd x
-\sqrt{\frac{n+1}{2}}\int_{-\infty}^{+\infty}\psi_n^*(x)\psi_{n+1}(x)\, \dd x
\right]=0.
\]

又
\[
\hat{p}^2=-\hbar^2 \dv[2]{}{x},
\]
从而
\begin{align*}
  \overline{p^2}
&=
\int_{-\infty}^{+\infty}\psi_n^*(x)\hat{p}^2\psi_n(x)\, \dd x
=
-\hbar^2\int_{-\infty}^{+\infty}\psi_n^*(x)\dv[2]{\psi_n}{x}\, \dd x\\
&=-\frac{\hbar^2\alpha^2}{2}\left[
\sqrt{n(n-1)}\int_{-\infty}^{+\infty}\psi_n^*(x)\psi_{n-2}(x)\, \dd x \right.\\
&\qquad \qquad-(2n+1)\int_{-\infty}^{+\infty}\psi_n^*(x)\psi_n(x)\, \dd x \\
&\qquad \qquad \left.+\sqrt{(n+1)(n+2)}\int_{-\infty}^{+\infty}\psi_n^*(x)\psi_{n+2}(x)\, \dd x\right]\\
&=\frac{\hbar^2\alpha^2}{2}(2n+1)\\
&=\frac{\hbar m\omega}{2}(2n+1).
\end{align*}

因此
\[
\overline{T}
=
\frac{\overline{p^2}}{2m}
=
\frac{\hbar\omega}{4}(2n+1).
\]
而
\[
E_n=\left( n+\frac{1}{2} \right)\hbar\omega=\frac{2n+1}{2}\hbar\omega,
\]
故
\[
\overline{T}=\frac{E_n}{2}.
\]

因此
\[
\overline{p}=0,
\qquad
\overline{T}=\frac{\overline{p^2}}{2m}=\frac{E_n}{2}.
\]

谐振子处于 $\psi_n$ 态下，计算
\[
\Delta x=\left[\overline{(x-\overline{x})^2}\right]^{1/2},\qquad
\Delta p=\left[\overline{(p-\overline{p})^2}\right]^{1/2},\qquad
\Delta x\Delta p=?
\]

\vspace{0.5cm}

\noindent\textbf{解：}
由前两题结果，在 \(\psi_n\) 态下，
\[
\overline{x}=0,
\qquad
\overline{p}=0,
\]
\[
\overline{x^2}=\frac{2n+1}{2\alpha^2},
\qquad
\overline{p^2}=\frac{\hbar^2\alpha^2}{2}\left(2n+1\right).
\]

因此
\[
\Delta x=\left[\overline{\left(x-\overline{x}\right)^2}\right]^{1/2}
=\left(\overline{x^2}-\overline{x}^{\,2}\right)^{1/2}

=\frac{1}{\alpha}\sqrt{\frac{2n+1}{2}}.

同理
\[

=\sqrt{\overline{p^2}}
=\hbar\alpha\sqrt{\frac{2n+1}{2}}.
\]
故
\[
\Delta x\Delta p
=
\frac{1}{\alpha}\sqrt{\frac{2n+1}{2}}
\cdot
\hbar\alpha\sqrt{\frac{2n+1}{2}}

\frac{2n+1}{2}\hbar.

\section{}
荷电 $q$ 的谐振子，受到外电场 $\mathcal{E}$ 的作用，
\[
V(x)=\frac{1}{2}m\omega^2x^2-q\mathcal{E}x
\]
求能量本征值和本征函数。（提示：对 $V(x)$ 进行配方，相当于谐振子势的平衡点不在 $x=0$ 点。）

\vspace{0.5cm}

\noindent\textbf{解：}
势能可配方为
\[
V(x)
=
\frac{1}{2}m\omega^2x^2-q\mathcal{E}x
=
\frac{1}{2}m\omega^2\left(x-\frac{q\mathcal{E}}{m\omega^2}\right)^2
-\frac{q^2\mathcal{E}^2}{2m\omega^2}.
\]

记
\[
x_0=\frac{q\mathcal{E}}{m\omega^2},
\]
则
\[
V(x)=\frac{1}{2}m\omega^2\left(x-x_0\right)^2-\frac{q^2\mathcal{E}^2}{2m\omega^2}.
\]

定态薛定谔方程为
\[
\left[
-\frac{\hbar^2}{2m}\dv[2]{}{x}
+\frac{1}{2}m\omega^2\left(x-x_0\right)^2
-\frac{q^2\mathcal{E}^2}{2m\omega^2}
\right]\psi(x)
=
E\psi(x).
\]

令
\[
\xi=x-x_0,
\]
则方程化为
\[
\left[
-\frac{\hbar^2}{2m}\dv[2]{}{\xi}
+\frac{1}{2}m\omega^2\xi^2
\right]\psi(\xi)
=
\left(
E+\frac{q^2\mathcal{E}^2}{2m\omega^2}
\right)\psi(\xi).
\]

这就是以 \(\xi=0\) 为平衡位置的标准谐振子方程，因此
\[
E+\frac{q^2\mathcal{E}^2}{2m\omega^2}
=
\left(n+\frac{1}{2}\right)\hbar\omega,
\qquad
n=0,1,2,\dots
\]
故能量本征值为
\[
E_n
=
\left(n+\frac{1}{2}\right)\hbar\omega
-\frac{q^2\mathcal{E}^2}{2m\omega^2},
\qquad
n=0,1,2,\dots
\]

本征函数就是谐振子本征函数整体平移 \(x_0\) 后的结果。记
\[
\alpha=\sqrt{\frac{m\omega}{\hbar}},
\]
则本征函数为
\[
\psi_n(x)
=
\left(\frac{\alpha}{\sqrt{\pi}\,2^n n!}\right)^{1/2}
H_n\left[\alpha\left(x-\frac{q\mathcal{E}}{m\omega^2}\right)\right]
\exp\left[
-\frac{\alpha^2}{2}
\left(x-\frac{q\mathcal{E}}{m\omega^2}\right)^2
\right].
\]

\section{}
一维无限深方势阱中的粒子，设初始时刻 $(t=0)$ 处于
\[
\psi(x,0)=\frac{\psi_1(x)+\psi_2(x)}{\sqrt{2}},
\]
$\psi_1(x)$ 与 $\psi_2(x)$ 分别为基态和第一激发态。求
\begin{enumerate}[label=(\alph*)]
  \item $\psi(x,t)$，$\rho(x,t)=\psi^*(x,t)\psi(x,t)$；
  \item 能量平均值 $\overline{H}$；
  \item 能量平方平均值 $\overline{H^2}$；
  \item 能量的涨落
  \[
  \Delta E=\left[\overline{(H-\overline{H})^2}\right]^{1/2};
  \]
  \item 体系的特征时间 $\Delta t$（分析 $\rho(x,t)$ 随时间变化的周期）。计算 $\Delta E\cdot\Delta t$。
\end{enumerate}


\vspace{0.5cm}

\noindent\textbf{解：}
一维无限深方势阱的能量本征态满足
\[
\hat{H}\psi_n(x)=E_n\psi_n(x),
\qquad
E_n=\frac{n^2\pi^2\hbar^2}{2ma^2}.
\]
因此
\[
E_1=\frac{\pi^2\hbar^2}{2ma^2},
\qquad
E_2=\frac{4\pi^2\hbar^2}{2ma^2}=4E_1.
\]

\begin{enumerate}[label=(\alph*)]
  \item 由定态叠加的时间演化，
\[
\psi(x,t)=\frac{1}{\sqrt{2}}\left[\psi_1(x)e^{-iE_1t/\hbar}+\psi_2(x)e^{-iE_2t/\hbar}\right].
\]
于是
\[
\psi^*(x,t)=\frac{1}{\sqrt{2}}\left[\psi_1(x)e^{iE_1t/\hbar}+\psi_2(x)e^{iE_2t/\hbar}\right].
\]
故概率密度为
\[
\rho(x,t)=\psi^*(x,t)\psi(x,t).
\]
\[
\rho(x,t)=\frac{1}{2}\left[\psi_1^2(x)+\psi_2^2(x)+2\psi_1(x)\psi_2(x)\cos\left(\frac{E_2-E_1}{\hbar}t\right)\right].
\]

  \item 由于 \(\psi_1,\psi_2\) 是哈密顿量本征态且彼此正交，
\[
\overline{H}=\ev{H}
=\matrixel{\psi}{\hat{H}}{\psi}
=\frac{1}{2}\left(E_1+E_2\right)
=\frac{5\pi^2\hbar^2}{4ma^2}.
\]

  \item 同理，
\[
\overline{H^2}=\ev{H^2}
=\frac{1}{2}\left(E_1^2+E_2^2\right)
=\frac{17}{2}E_1^2.
\]

  \item 能量涨落为
\begin{align*}
  \Delta E&=\left[\overline{\left(H-\overline{H}\right)^2}\right]^{1/2}
  =\left(\overline{H^2}-\overline{H}^{\,2}\right)^{1/2}\\
  &=\left[
  \frac{1}{2}\left(E_1^2+E_2^2\right)
  -\frac{1}{4}\left(E_1+E_2\right)^2
  \right]^{1/2}\\
  &=\frac{1}{2}\abs{E_2-E_1}=3E_1=\frac{3\pi^2\hbar^2}{4ma^2}.
\end{align*}

  \item \(\rho(x,t)\) 随时间的变化只出现在因子
\[
\cos\left(\frac{E_2-E_1}{\hbar}t\right)
\]
中，因此其周期为
\[
\Delta t=\frac{2\pi\hbar}{E_2-E_1}.
\]
由于
\[
E_2-E_1=3E_1=\frac{3\pi^2\hbar^2}{2ma^2},
\]
故
\[
\Delta t=\frac{2\pi\hbar}{3E_1}
=\frac{4ma^2}{3\pi\hbar}.
\]
因此
\[
\Delta E\cdot \Delta t
=
\frac{1}{2}\left(E_2-E_1\right)\cdot \frac{2\pi\hbar}{E_2-E_1}
=
\pi\hbar.
\]
\end{enumerate}

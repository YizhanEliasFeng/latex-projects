\section{}
设质量为 $m$ 的粒子在势场 $V(\vb{r})$ 中运动。

\begin{enumerate}[label=(\alph*)]
    \item 证明粒子的能量平均值为
    \[
    E = \int W\, \dd^3r,
    \]
    式中
    \[
    W = \frac{\hbar^2}{2m} \nabla \psi^* \cdot \nabla \psi + \psi^* V \psi
    \qquad \text{(能量密度)}
    \]

    \item 证明能量守恒式
    \[
    \pdv{W}{t} + \nabla \cdot \vb{s} = 0,
    \]
    \[
    \vb{s} = -\frac{\hbar^2}{2m}
    \left(
    \pdv{\psi^*}{t}\nabla \psi + \pdv{\psi}{t}\nabla \psi^*
    \right)
    \qquad \text{(能流密度)}
    \]
\end{enumerate}

\vspace{0.5cm}

\noindent\textbf{证明：} 
\begin{enumerate}[label=(\alph*)]
    \item 设 $\psi$ 已归一化，则粒子能量平均值为
    \begin{align*}
      \overline{E} &= \int \psi^*\left(-\frac{\hbar^2}{2m}\nabla^2 + V\right)\psi\, \dd^3r \\
      &= \int \psi^* V \psi\, \dd^3r + \int \psi^*\left(-\frac{\hbar^2}{2m}\nabla^2\right)\psi\, \dd^3r \\
      &= \int \psi^* V \psi\, \dd^3r -\frac{\hbar^2}{2m}\int \left[\nabla\cdot(\psi^*\nabla\psi) - (\nabla\psi^*)\cdot(\nabla\psi)\right] d^3r \\
      &= \int \psi^* V \psi\, \dd^3r + \frac{\hbar^2}{2m}\int \nabla\psi^*\cdot\nabla\psi\, \dd^3r \\
      &=\int \left(\frac{\hbar^2}{2m}\nabla\psi^*\cdot\nabla\psi + \psi^*V\psi\right)d^3r \\
      &= \int W\, \dd^3r.
    \end{align*}

    \item 
    \begin{align*}
      \pdv{W}{t} &= \frac{\hbar^2}{2m}\left(\nabla \dot{\psi}^* \cdot \nabla \psi + \nabla \psi^* \cdot \nabla \dot{\psi}\right) + \dot{\psi}^* V\psi + \psi^* V\dot{\psi} \\
      &= \frac{\hbar^2}{2m}\left[\nabla\cdot(\dot{\psi}^*\nabla\psi + \dot{\psi}\nabla\psi^*) - (\dot{\psi}^*\nabla^2\psi + \dot{\psi}\nabla^2\psi^*)\right] + \dot{\psi}^*V\psi + \psi^*V\dot{\psi} \\
      &= -\nabla\cdot\vb{s} + \dot{\psi}^*\left(-\frac{\hbar^2}{2m}\nabla^2 + V\right)\psi + \dot{\psi}\left(-\frac{\hbar^2}{2m}\nabla^2 + V\right)\psi^*.
    \end{align*}
    由 Schr\"odinger 方程及其共轭式可知上式后两项互相抵消，因此
    \[
    \pdv{W}{t} + \nabla\cdot\vb{s} = 0.
    \]
\end{enumerate}

\section{}
考虑单粒子的 Schr\"odinger 方程
\[
 i\hbar \pdv{\psi(\vb{r},t)}{t}
 = -\frac{\hbar^2}{2m}\nabla^2\psi(\vb{r},t)
 + [V_1(\vb{r})+iV_2(\vb{r})]\psi(\vb{r},t)
\]
$V_1$ 与 $V_2$ 为实函数。

\begin{enumerate}[label=(\alph*)]
    \item 证明粒子的概率（粒子数）不守恒；
    \item 证明粒子在空间体积 $\tau$ 内的概率随时间的变化为
    \[
    \dv{}{t}\iiint_\tau \dd^3r\, \psi^*\psi
    = -\frac{\hbar}{2im}\iint_S (\psi^*\nabla\psi - \psi\nabla\psi^*)\cdot \dd\vb{S}
    + \frac{2}{\hbar}\iiint_\tau V_2(\vb{r})\, \dd^3r\, \psi^*\psi.
    \]
\end{enumerate}

\vspace{0.5cm}

\noindent\textbf{证明：}取 Schr\"odinger 方程
\[
 i\hbar \pdv{\psi}{t} = -\frac{\hbar^2}{2m}\nabla^2\psi + [V_1+iV_2]\psi \tag{1}
\]
的复共轭
\[
 -i\hbar \pdv{\psi^*}{t} = -\frac{\hbar^2}{2m}\nabla^2\psi^* + [V_1-iV_2]\psi^* \tag{2}
\]
$(1)\times \psi^* - (2)\times \psi$ 得
\begin{align*}
   i\hbar \pdv{}{t}(\psi^*\psi)
   &= -\frac{\hbar^2}{2m}(\psi^*\nabla^2\psi - \psi\nabla^2\psi^*) + 2iV_2\psi^*\psi\\
   &= -\frac{\hbar^2}{2m}\nabla\cdot(\psi^*\nabla\psi - \psi\nabla\psi^*) + 2iV_2\psi^*\psi.
\end{align*}
两边同时积分，并利用 Stokes 定理，得
\[
 \pdv{}{t}\int_\tau \dd^3r\, \psi^*\psi
 = -\frac{\hbar}{2im}\int_S \dd\vb{S}\cdot(\psi^*\nabla\psi - \psi\nabla\psi^*)
 + \frac{2}{\hbar}\int_\tau \dd^3r\, V_2\psi^*\psi.
\]
当 $\tau\to\infty$ 时，上式第一项为 $0$，而 $V_2\neq 0$，所以
\[
 \pdv{}{t}\int_\tau d^3r\, \psi^*\psi \neq 0,
\]
即粒子数不守恒。

\section{}
对于一维自由粒子，

\begin{enumerate}[label=(\alph*)]
    \item 设波函数为
    \[
    \psi_p(x)=\frac{1}{\sqrt{2\pi\hbar}}e^{ipx/\hbar},
    \]
    试用 Hamilton 算符
    \[
    H = \frac{\hat p^2}{2m} = -\frac{\hbar^2}{2m}\dv[2]{}{x}
    \]
    对 $\psi_p(x)$ 运算，验证 $H\psi_p(x)=\dfrac{p^2}{2m}\psi_p(x)$；说明动量本征态 $\psi_p(x)$ 是 Hamilton 量（能量）本征态，能量本征值为 $E=p^2/2m$；

    \item 设粒子在初始 $(t=0)$ 时刻，$\psi(x,0)=\psi_{p}(x)$，求 $\psi(x,t)$；

    \item 设波函数为
    \[
    \psi(x)=\delta(x)=\frac{1}{\sqrt{2\pi}}\int e^{ikx}\, \dd k
    = \frac{1}{\sqrt{2\pi\hbar}}\int e^{ipx/\hbar}\, \dd p,
    \]
    可以看成无穷多个平面波 $e^{ikx}$ 的叠加，即无穷多个动量本征态 $e^{ipx/\hbar}$ 的叠加。试问 $\psi(x)=\delta(x)$ 是否是能量本征态？

    \item 设粒子在 $t=0$ 时刻 $\psi(x,0)=\delta(x)$，求 $\psi(x,t)$。
\end{enumerate}

\vspace{0.5cm}

\noindent\textbf{解}
\begin{enumerate}[label=(\alph*)]
  \item
  \[
  H\psi_p = -\frac{\hbar^2}{2m}\dv[2]{}{x}\left[\frac{1}{\sqrt{2\pi\hbar}}e^{ipx/\hbar}\right]
  = \frac{p^2}{2m}\psi_p.
  \]
  所以动量本征态 $\psi_p(x)$ 是 Hamilton 量的本征态，能量本征值为
  \[
  E = \frac{p^2}{2m}.
  \]

  \item $\psi(x,0)=e^{ip_0x/\hbar}$，其 Fourier 变换为
  \[
  \varphi(p)=\frac{1}{\sqrt{2\pi\hbar}}\int e^{ip_0x/\hbar}e^{-ipx/\hbar}\, \dd x
  = \sqrt{2\pi\hbar}\,\delta(p-p_0).
  \]
  由于 $\psi(x,0)$ 是能量本征态，按教材公式可得
  \[
  \psi(x,t)=\frac{1}{\sqrt{2\pi\hbar}}\int_{-\infty}^{+\infty}\sqrt{2\pi\hbar}
  e^{ipx/\hbar}\delta(p-p_0)e^{-iEt/\hbar}\, \dd p
  = e^{ip_0x/\hbar - iEt/\hbar}.
  \]

  \item 对于自由粒子，动量本征态即为能量本征态。由于 $\delta(x)$ 是无穷多个动量本征态 $e^{ipx/\hbar}$ 的叠加，所以 $\psi(x)=\delta(x)$ 不是能量本征态。

  \item 因为 $\psi(x,0)=\delta(x)$，按教材公式
  \[
  \varphi(p)=\frac{1}{\sqrt{2\pi\hbar}}\int_{-\infty}^{+\infty}\delta(x)e^{-ipx/\hbar}\, \dd x=(2\pi\hbar)^{-1/2}.
  \]
  所以
  \begin{align*}
  \psi(x,t)&=\frac{1}{2\pi\hbar}\int_{-\infty}^{+\infty}e^{i(px-Et)/\hbar}\, \dd p\\
  &= \frac{1}{2\pi\hbar}\int_{-\infty}^{+\infty}\exp\left[\frac{i}{\hbar}\left(px-\frac{p^2}{2m}t\right)\right]\dd p\\
  &= \sqrt{\frac{m}{2\pi\hbar t}}\, e^{-i\pi/4} e^{i\frac{mx^2}{2\hbar t}}.
  \end{align*}  
\end{enumerate}

\section{}
设一维自由粒子的初态为一个 Gauss 波包，
\[
\psi(x,0)=e^{ip_0x/\hbar}\frac{1}{(\pi a^2)^{1/4}}e^{-x^2/2a^2}.
\]

\begin{enumerate}[label=(\arabic*)]
    \item 证明初始时刻，$\overline x = 0$，$\overline p = p_0$，
      \[
      \Delta x = \left[\overline{(x-\overline x)^2}\right]^{1/2}=\frac{a}{\sqrt{2}},
      \]
      \[
      \Delta p = \left[\overline{(p-\overline p)^2}\right]^{1/2}=\frac{\hbar}{\sqrt{2}a},
      \]
      \[
      \Delta x\cdot \Delta p = \frac{\hbar}{2}.
      \]
    \item 计算 $t$ 时刻的波函数。
\end{enumerate}

\vspace{0.5cm}

\noindent\textbf{解}
\begin{enumerate}[label=(\arabic*)]
  \item 初始时刻
  \[
  \overline x = \int_{-\infty}^{+\infty} x\abs{\psi}^2\, \dd x
  = \int_{-\infty}^{+\infty} x\frac{1}{\sqrt{\pi a^2}}e^{-x^2/a^2}\, \dd x = 0
  \]
  \[
  \overline{x^2} = \int_{-\infty}^{+\infty}x^2\abs{\psi}^2\, \dd x
  = \int_{-\infty}^{+\infty}x^2\frac{1}{\sqrt{\pi a^2}}e^{-x^2/a^2}\, \dd x = \frac{a^2}{2}
  \]
  \[
  \Delta x = (\overline{x^2}-\overline x^2)^{1/2}=\frac{a}{\sqrt{2}}.
  \]
  按 Fourier 逆变换，
  \begin{align*}
    \varphi(p)&=\frac{1}{\sqrt{2\pi\hbar}}\int_{-\infty}^{+\infty}\psi(x,0)e^{-ipx/\hbar}\, \dd x \\
    &= \frac{1}{\sqrt{2\pi\hbar}}\int_{-\infty}^{+\infty}e^{ip_0x/\hbar}\frac{1}{(\pi a^2)^{1/4}}e^{-x^2/2a^2}e^{-ipx/\hbar}\, \dd x \\
    &= \left(\frac{a^2}{\pi\hbar^2}\right)^{1/4}
    e^{-\frac{a^2(p-p_0)^2}{2\hbar^2}}.
  \end{align*}
  因而
  \[
  \abs{\varphi(p)}^2 = \sqrt{\frac{a^2}{\pi\hbar^2}}e^{-\frac{a^2(p-p_0)^2}{\hbar^2}}.
  \]
  所以
  \[
  \overline p = \int_{-\infty}^{+\infty}p\abs{\varphi(p)}^2\, \dd p = p_0,
  \]
  \[
  \overline{p^2} = \int_{-\infty}^{+\infty}p^2\abs{\varphi(p)}^2\, \dd p = p_0^2 + \frac{\hbar^2}{2a^2},
  \]
  \[
  \Delta p = (\overline{p^2}-\overline p^2)^{1/2}=\frac{\hbar}{\sqrt{2}a}.
  \]
  因此
  \[
  \Delta x\cdot\Delta p = \frac{\hbar}{2}.
  \]

  \item 按教材讨论可知，在 $t>0$ 时的波函数为
  \begin{align*}
    \psi(x,t)&=\frac{1}{\sqrt{2\pi\hbar}}\int \varphi(p)
    \exp\left[\frac{i}{\hbar}\left(px-\frac{p^2}{2m}t\right)\right]\dd p \\
    & = [\sqrt{\pi}(a+\frac{i\hbar t}{ma})]^{-1/2}
    \exp\left[\frac{i p_0}{\hbar}\left(x-\frac{p_0 t}{m}\right)
    - \frac{\left(x-p_0 t/m\right)^2}{2a^2\left(1+i\hbar t/ma^2\right)}\right].
  \end{align*}
\end{enumerate}


\section{}
设一维自由粒子的初态为 $\psi(x,0)$，证明在足够长时间后，
\[
\psi(x,t)=\sqrt{\frac{m}{\hbar t}}\exp(-i\pi/4)\cdot
\exp\left(\frac{imx^2}{2\hbar t}\right)\cdot
\varphi\left(\frac{mx}{\hbar t}\right)
\]
式中
\[
\varphi(k)=\frac{1}{\sqrt{2\pi}}\int_{-\infty}^{+\infty}\psi(x,0)e^{-ikx}\, \dd x
\]
是 $\psi(x,0)$ 的 Fourier 变换。

\noindent\textbf{提示：}
利用
\[
\lim_{a\to\infty}\sqrt{\frac{a}{\pi}}e^{i\pi/4}e^{-iax^2}=\delta(x).
\]

\vspace{0.5cm}

\noindent\textbf{证明：}根据自由粒子的动量本征态随时间变化的规律$e^{ikx}\to e^{-i(\hbar k^2/2m)t}e^{ikx}$，所以时刻 $t$ 的波函数为
\[
\psi(x,t)=\frac{1}{\sqrt{2\pi}}\int_{-\infty}^{+\infty}\varphi(k)e^{ikx-t\hbar k^2/2mi}\, \dd k.
\]
当时间足够长后（$t\to\infty$），由提示可知上式被积函数中指数函数具有 $\delta$ 函数的性质，即
\[
\psi(x,t)\approx \frac{1}{\sqrt{2\pi}}e^{imx^2/2\hbar t}
\int \sqrt{\frac{2\pi m}{\hbar t}}e^{-i\pi/4}\varphi(k)
\delta\left(k-\frac{mx}{\hbar t}\right)\dd k
\]
\[
= \sqrt{\frac{m}{\hbar t}}e^{-i\pi/4}e^{imx^2/(2\hbar t)}
\varphi\left(\frac{mx}{\hbar t}\right).
\]

\section{}
按照粒子密度分布 $\rho$ 和粒子流密度分布 $\vb{j}$ 的表示式
\[
\rho(\vb{r},t)=\psi^*(\vb{r},t)\psi(\vb{r},t)
\]
\[
\vb{j}(\vb{r},t)=-\frac{i\hbar}{2m}\left[\psi^*(\vb{r},t)\nabla\psi(\vb{r},t)-\psi(\vb{r},t)\nabla\psi^*(\vb{r},t)\right]
\]
定义粒子的速度分布 $\vb{v}$
\[
\vb{v} = \frac{\vb{j}}{\rho}
= -\frac{i\hbar}{2m}\left[\frac{\nabla\psi(\vb{r},t)}{\psi(\vb{r},t)} - \frac{\nabla\psi^*(\vb{r},t)}{\psi^*(\vb{r},t)}\right].
\]
证明
\[
\nabla\times \vb{v} = 0.
\]
设想 $\vb{v}$ 描述一个速度场，则 $\vb{v}$ 为一个无旋场。

\vspace{0.5cm}

\noindent\textbf{证明：}按照上述定义，可知
\[
\vb{v}
= -\frac{i\hbar}{2m}\left[\frac{\nabla\psi}{\psi} - \frac{\nabla\psi^*}{\psi^*}\right]
= -\frac{i\hbar}{2m}[\nabla\ln\psi - \nabla\ln\psi^*]
= -\frac{i\hbar}{2m}\nabla\ln\frac{\psi}{\psi^*}.
\]
于是
\[
\nabla\times \vb{v}
= -\frac{i\hbar}{2m}\nabla\times\nabla\ln\frac{\psi(\vb{r},t)}{\psi^*(\vb{r},t)} = 0.
\]

\section{}
处于势场 $V(\vb{r})$ 中的粒子，在坐标表象中的能量本征方程表示成
\[
\left[-\frac{\hbar^2}{2m}\nabla^2 + V(\vb{r})\right]\psi(\vb{r})=E\psi(\vb{r})
\]
试在动量表象中写出相应的能量本征方程。

\vspace{0.5cm}

\noindent\textbf{解：}
利用 $\psi(\vb{r})$ 的 Fourier 变换
\[
\psi(\vb{r})=\frac{1}{(2\pi\hbar)^{3/2}}\int \varphi(\vb{p})e^{i\vb{p}\cdot \vb{r}/\hbar}\, \dd^3 p
\]
可知
\begin{align*}
  \left[-\frac{\hbar^2}{2m}\nabla^2 + V(\vb{r})\right]\psi(\vb{r})
  &= \left(-\frac{\hbar^2}{2m}\nabla^2 + V(\vb{r})\right)
  \frac{1}{(2\pi\hbar)^{3/2}}\int \varphi(\vb{p})e^{i\vb{p}\cdot\vb{r}/\hbar}\, \dd^3p \\
  &= \frac{1}{(2\pi\hbar)^{3/2}}\left[\int \left(\frac{\vb{p}^2}{2m} + V\left(i\hbar\pdv{}{\vb{p}}\right)\right)
  \varphi(\vb{p})e^{i\vb{p}\cdot\vb{r}/\hbar}\, \dd^3p\right]\\
  &= \frac{1}{(2\pi\hbar)^{3/2}}\int E\varphi(\vb{p})e^{i\vb{p}\cdot\vb{r}/\hbar}\, \dd^3p.
\end{align*}
即
\[
\frac{1}{(2\pi\hbar)^{3/2}}
\int \left[\frac{\vb{p}^2}{2m}+V\left(i\hbar\pdv{}{\vb{p}}\right)-E\right]
\varphi(\vb{p})e^{i\vb{p}\cdot\vb{r}/\hbar}\, \dd^3p = 0.
\]
所以在动量表象中相应的能量本征方程为
\[
\left[\frac{\vb{p}^2}{2m}+V\left(i\hbar\pdv{}{\vb{p}}\right)\right]\varphi(\vb{p})=E\varphi(\vb{p}).
\]
